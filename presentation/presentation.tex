\section{Context \& Motivation}

% ==============================================================================
% SLIDE 2: CONTESTO & MOTIVAZIONE
% ==============================================================================
\begin{frame}{Context: Software Evolution \& Dependency Analysis}{Technical Debt and the Role of Dependency Graphs}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \textbf{Software Evolution \& Degradation}\par\vspace{0.2em}
      {\small Continuous feature delivery and refactoring degrade code structure over time, accumulating architectural technical debt.}\par\vspace{0.8em}

      \textbf{The Polyglot Landscape}\par\vspace{0.2em}
      {\small Modern enterprise systems routinely combine multiple languages and runtime environments, breaking single-language assumptions.}\par\vspace{0.8em}

      \textbf{The Core Problem}\par\vspace{0.2em}
      {\small Detecting structural degradation requires extracting accurate dependency graphs across diverse, heterogeneous codebases.}

    \column{0.48\textwidth}
      \begin{cookiecard}[title={Why Extract Dependency Graphs?}]
        \fontsize{8.0}{10.5}\selectfont
        \textbf{Architectural Smell Detection}\par
        Identify cyclical dependencies, unstable interfaces, hub-like modules, and god components.\par\vspace{0.45em}
        
        \textbf{Impact Analysis \& Refactoring}\par
        Trace ripple effects and blast radius across modules before applying architectural changes.\par\vspace{0.45em}
        
        \textbf{System Comprehension}\par
        Provide an objective, queryable structural topology of enterprise software assets.
      \end{cookiecard}
  \end{columns}
\end{frame}

% ==============================================================================
% SLIDE 3: STATO DELL'ARTE & FRAMMENTAZIONE
% ==============================================================================
\begin{frame}{State of the Art: Tooling Fragmentation}{The Cost of Language-Specific Analysis Pipelines}
  \begin{columns}[c,onlytextwidth]
    \column{0.48\textwidth}
      \begin{itemize}\setlength{\itemsep}{0.30em}
        \item \textbf{Tightly Coupled to Compilers:}\\
          {\small Analyzers depend on specific compiler front-ends and heavy runtimes (\texttt{javac}, Clang, JVM).}
          
        \item \textbf{Duplicated Engineering:}\\
          {\small AST traversal, scope resolution, and graph emission are re-implemented from scratch for every language.}
          
        \item \textbf{Heavyweight Multi-Language Tools:}\\
          {\small Supporting multiple languages forces tools to ship and maintain all underlying compiler toolchains.}
          
        \item \textbf{Tooling Fragmentation:}\\
          {\small High engineering barrier leads to fragmented, single-language tools and slow adoption.}
      \end{itemize}

    \column{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.44\textheight,keepaspectratio]{../images/background/mxn_problem.png}
      \vspace{0.2em}
      \captionof{figure}{\scriptsize Tool proliferation ($M \times N$) vs unified intermediate representation ($M + N$).}
  \end{columns}
\end{frame}

\section{The OmniDeps Framework}

% ==============================================================================
% SLIDE 6: ARCHITETTURA DELLA PIPELINE
% ==============================================================================
\begin{frame}[t]{Analysis Pipeline Overview}{Functional Composition and Type-Safe Staging}
  \vspace{-0.2em}
  \centering
  {\small $\displaystyle \phi : \mathcal{W} \to \mathcal{G}, \qquad \phi = \gamma \circ \rho \circ \varepsilon = \gamma \circ (\rho_{\text{exec}} \circ \rho_{\text{build}}) \circ \varepsilon$}\par
  \vspace{0.5em}

  \centering
  \includegraphics[width=\textwidth,keepaspectratio]{../images/analysis_process/pipeline_detailed.png}

  \begin{columns}[T,onlytextwidth]
    \column{0.32\textwidth}
      {\fontsize{7.5}{9.2}\selectfont
      $\varepsilon : \mathcal{W} \to \mathcal{D}$\par
      Parses raw CSTs via Tree-sitter and applies heuristic extraction into unlinked IR. Only phase aware of native syntax.
      }

    \column{0.34\textwidth}
      {\fontsize{7.5}{9.2}\selectfont
      $\rho : \mathcal{D} \to \mathcal{D}_{\text{res}}$\par
      Builds Scope Tree ($\rho_{\text{build}}$) and resolves symbolic identifiers ($\rho_{\text{exec}}$) via algebraic scope climbing. Purely logical.
      }

    \column{0.31\textwidth}
      {\fontsize{7.5}{9.2}\selectfont
      $\gamma : \mathcal{D}_{\text{res}} \to \mathcal{G}$\par
      Flattens resolved entities into component vertices ($V$) and emits directed dependency edges ($E$). Purely structural.
      }
  \end{columns}


\end{frame}

% ==============================================================================
% SLIDE 7: FASE 1 - EXTRACTION
% ==============================================================================
\begin{frame}{Phase 1: Syntactic Extraction}{From Concrete Syntax Trees to Intermediate Representation}
  \begin{columns}[T,onlytextwidth]
    \column{0.32\textwidth}
      \cookiekicker{CST Traversal}\par\vspace{0.2em}
      {\fontsize{7.5}{9.2}\selectfont
      Parses source files into Concrete Syntax Trees using Tree-sitter. Systematically traverses the tree to inspect syntactic constructs while abstracting lexical noise.
      }

    \column{0.34\textwidth}
      \cookiekicker{Heuristic Recognition}\par\vspace{0.2em}
      {\fontsize{7.5}{9.2}\selectfont
      Applies declarative heuristics on CST nodes to identify architectural constructs. Directly leverages \textbf{structural convergence} to recognize equivalent patterns across languages.
      }

    \column{0.31\textwidth}
      \cookiekicker{IR Instantiation}\par\vspace{0.2em}
      {\fontsize{7.5}{9.2}\selectfont
      Extracts relevant identifiers, types, and references from source buffers, populating the unlinked universal Intermediate Representation for subsequent analysis.
      }
  \end{columns}

  \vspace{0.45em}
  \centering
  \includegraphics[width=0.96\textwidth,height=0.33\textheight,keepaspectratio]{../images/extraction/extraction_layers.png}
\end{frame}

      \cookiekicker{Conservative Extension}\par\vspace{0.15em}
      {\fontsize{7.5}{9.0}\selectfont

    \column{0.34\textwidth}

      }

      \cookiekicker{Unified Abstraction}\par\vspace{0.15em}
      {\fontsize{7.5}{9.0}\selectfont
      Normalizes diverse constructs (classes, traits, namespaces) into a uniform model isolating subsequent phases from syntax.
      }

\begin{frame}{Phase 2: Decoupled Name Resolution}{TypeRef Binding and Two-Step Scope Navigation}
  \begin{columns}[c,onlytextwidth]
    \column{0.35\textwidth}
      \cookiekicker{TypeRef Resolution}\par\vspace{0.15em}
      {\fontsize{7.5}{9.2}\selectfont
      Symbol references (variable types, calls) start as unresolved \texttt{TypeRef}s. Phase 2 binds them to concrete targets.\par}
      \vspace{0.45em}

      \cookiekicker{Two-Step Decoupling}\par\vspace{0.15em}
      {\fontsize{7.5}{9.2}\selectfont
      \textbf{1. Local Query Building:}\\
      Local symbol stack rewrites references into closed queries, then frees local state.\par\vspace{0.20em}
      \textbf{2. Global Query Execution:}\\
      Evaluates queries deterministically over the global Scope Tree without file-order bias.\par}
      \vspace{0.45em}

      \cookiekicker{Three Query Primitives}\par\vspace{0.20em}
      \cookiebadge{Find} {\fontsize{7.0}{8.5}\selectfont Lexical climbing}\hspace{0.2em}\\
      \cookiebadge{Extract} {\fontsize{7.0}{8.5}\selectfont Member descent}\hspace{0.2em}\\
      \cookiebadge{Call} {\fontsize{7.0}{8.5}\selectfont Return propagation}\\

\begin{frame}{Discussion: Architectural Trade-offs}{Compiler-Coupled Precision vs. Lightweight Polyglot Scalability}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
      \begin{cookiecard}[title={Demonstrated Advantages}]
        {\fontsize{7.0}{8.6}\selectfont
        \begin{itemize}\setlength{\itemsep}{0.22em}\setlength{\leftskip}{-1.2em}
          \item \textbf{Zero Build Configuration:} Operates directly on source code without host SDKs, compiler toolchains, or build scripts.
          \item \textbf{Polyglot Extensibility:} Single unified pipeline; adding a language requires only CST mapping heuristics in Phase 1.
          \item \textbf{High Throughput \& Low Memory:} Flat memory footprint; scans multi-million-line repositories in seconds.
          \item \textbf{Practical Resilience:} \textbf{99.4\% completion rate} across real-world software where compiler tools crash or timeout.
        \end{itemize}\par}
      \end{cookiecard}

    \column{0.49\textwidth}
      \begin{cookiecard}[title={Recognized Limitations}]
        {\fontsize{7.0}{8.6}\selectfont
        \begin{itemize}\setlength{\itemsep}{0.22em}\setlength{\leftskip}{-1.2em}
          \item \textbf{External Binary Boundaries:} Call chains traversing closed precompiled libraries cannot resolve intermediate return types.
          \item \textbf{Dynamic Ecosystems:} In dynamic languages (Python), call-site precision is fundamentally bounded by PEP 484 type hint adoption.
          \item \textbf{Separate Compilation:} C/C++ macro masking requires preprocessing, which replicates shared headers across translation units.
        \end{itemize}\par}
      \end{cookiecard}
  \end{columns}
  \vspace{0.45em}
  \centering
  {\fontsize{7.5}{9.0}\selectfont\textbf{Core Architectural Trade-off:} \textit{Exhaustive Compiler Verification} $\;\longleftrightarrow\;$ \textit{Lightweight Scalability for Continuous Auditing}\par}
\end{frame}
